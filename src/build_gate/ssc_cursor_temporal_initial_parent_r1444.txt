type FinalChordSourceRange = (usize, usize);
type FinalChordSourceDescriptor = (usize, usize, i64, Option<i32>);
type FinalChordSourceByRange =
    std::collections::BTreeMap<FinalChordSourceRange, FinalChordSourceDescriptor>;

#[derive(Debug, Clone)]
struct FinalChordSourceBindingRow {
    fragment_id: usize,
    measure: usize,
    raw_note_index: usize,
    source_onset: i64,
    source_pitch: Option<i32>,
    fragment: CandidateMaterialFragmentState,
}

fn final_chord_source_binding(
    cache: &mut std::collections::BTreeMap<(usize, usize), FinalChordSourceBindingRow>,
    source_range: (usize, usize),
    source_by_range: &FinalChordSourceByRange,
    fragment_by_raw: &std::collections::BTreeMap<(usize, usize), CandidateMaterialFragmentState>,
    fragment_id_by_raw: &std::collections::BTreeMap<(usize, usize), usize>,
    missing_source_message: &'static str,
    missing_fragment_message: &'static str,
) -> anyhow::Result<FinalChordSourceBindingRow> {
    if let Some(binding) = cache.get(&source_range) {
        return Ok(binding.clone());
    }
    let (measure, raw_note_index, source_onset, source_pitch) = source_by_range
        .get(&source_range)
        .copied()
        .ok_or_else(|| anyhow!("{}", missing_source_message))?;
    let fragment = fragment_by_raw
        .get(&(measure, raw_note_index))
        .cloned()
        .ok_or_else(|| anyhow!("{}", missing_fragment_message))?;
    let fragment_id = fragment_id_by_raw
        .get(&(measure, raw_note_index))
        .copied()
        .ok_or_else(|| {
            anyhow!(
                "FinalChordSourceBinding: CandidateMaterial fragment has no canonical fragment id"
            )
        })?;
    let binding = FinalChordSourceBindingRow {
        fragment_id,
        measure,
        raw_note_index,
        source_onset,
        source_pitch,
        fragment,
    };
    cache.insert(source_range, binding.clone());
    Ok(binding)
}

// Representation provenance binds occurrences, not necessarily new attacks.
// A tie continuation retains its existing attack identity at a later fragment
// onset. Only an actual onset change enters the attack translation owner.
fn final_chord_forward_attack_translation(
    binding: &FinalChordSourceBindingRow,
    attack: &CandidateMaterialAttackState,
    source_onset: i64,
    target_onset: i64,
) -> anyhow::Result<Option<CandidateMaterialForwardOnsetTranslation>> {
    let fragment = &binding.fragment;
    if binding.source_onset != source_onset
        || fragment.measure != binding.measure
        || fragment.raw_note_index != binding.raw_note_index
        || fragment.onset != source_onset
        || fragment.duration <= 0
        || fragment.pitch != binding.source_pitch
        || fragment.pitch != Some(attack.pitch)
        || fragment.source_event_id != Some(attack.source_event_id)
        || fragment.attack_id != Some(attack.attack_id)
        || !attack.exists
    {
        return Err(anyhow!(
            "FinalChordEncodingSoundingMaterialization: bound occurrence identity disagrees with CandidateMaterialState for SourceEventId {}",
            attack.source_event_id.0
        ));
    }
    if source_onset == target_onset && fragment.tie_stop {
        return Ok(None);
    }
    if attack.measure != binding.measure || attack.onset != source_onset {
        return Err(anyhow!(
            "FinalChordEncodingSoundingMaterialization: forward identity bridge disagrees with CandidateMaterialState for SourceEventId {}; fragment_id={} fragment_measure={} attack_measure={} source_onset={} attack_onset={} target_onset={} tie_stop={}",
            attack.source_event_id.0, binding.fragment_id, binding.measure, attack.measure,
            source_onset, attack.onset, target_onset, fragment.tie_stop,
        ));
    }
    if source_onset == target_onset {
        return Ok(None);
    }
    Ok(Some(CandidateMaterialForwardOnsetTranslation {
        source_event_id: attack.source_event_id,
        attack_id: attack.attack_id,
        target_onset,
    }))
}

            structural_write_authority::RenderSinkKind::Checkpoint(structural_write_authority::RenderCheckpointSink::FinalNotation),

        let render_preview_source = render_state.require_document("Condensation preview").to_owned();

            }

    // move-only with the export to the released-Master adoption.

        repair_cursor_lane_end_invariant_xml("MasterXML", best.layout_xml());

        crate::master_state_identity::RenderMasterEpoch::after_cursor_lane_before_final_chord(
            best.master_state.lineage(),
            best.master_state.document(),
            master_export_block_receipt.as_ref(),
            &final_export_xml,
        ),

    // no receipt remains to transport in InitialRenderTransaction.

            pipeline_line: format!(
                "ReanchorRestVisibilitySafetyNet v{}: übersprungen; keine finale Render-XML-Quelle; keine MusicXML-Änderung, MusicMutation nein.",

        let mut receipts = crate::master_state_identity::RenderTransitionReceipts::new();
        receipts.push_exact(receipt);

                structural_write_authority::RenderSinkKind::Checkpoint(structural_write_authority::RenderCheckpointSink::RetentionReturn),
                |path, _pre, post| crate::structural_write_authority::write_retention_return_checkpoint_authorized(path, post),
            ).with_context(|| "IR13 P2 retention adoption")?;
        } else { render_state.decline_transition(root, receipts); }
        audit
    } else {
        RetentionFirstP2AssertiveReturnWriteAudit {
            ..RetentionFirstP2AssertiveReturnWriteAudit::default()
        }
    };
    let mut upper_staff_final_duplicate_rest_rewrite_written = false;
    if final_render_source_available {
            let root = render_state.require_state_root("IR14 upper duplicate rest");
            let render_xml_before_upper_duplicate_rewrite = root.document().to_owned();
            let (before_events, before_backup_forward_ok) =
                collect_render_xml_note_events(&render_xml_before_upper_duplicate_rewrite);

                && upper_staff_final_duplicate_rest_rewrite_hidden_pitched_before
                    == upper_staff_final_duplicate_rest_rewrite_hidden_pitched_after
                && upper_staff_final_duplicate_rest_rewrite_hidden_rests_before
                    == upper_staff_final_duplicate_rest_rewrite_hidden_rests_after
                && upper_staff_final_duplicate_rest_rewrite_visible_rests_after
                    < upper_staff_final_duplicate_rest_rewrite_visible_rests_before;
            if invariant_ok {
                render_state.adopt_transition(root, &render_xml_after_upper_duplicate_rewrite, receipts,
                    structural_write_authority::RenderSinkKind::Checkpoint(structural_write_authority::RenderCheckpointSink::DuplicateRestRewrite),
                    |path, _pre, post| crate::structural_write_authority::write_duplicate_rest_rewrite_checkpoint_authorized(path, post),
                ).with_context(|| "IR14 duplicate rest adoption")?;
                upper_staff_final_duplicate_rest_rewrite_written = true;
            } else { render_state.decline_transition(root, receipts); }
        }
    let strict_staff_local_release_completeness_trial = if final_render_source_available {
        let root = render_state.require_state_root("IR15 StrictStaff");

                |path, pre, post| (crate::structural_write_authority::write_strict_staff_release_trial_authorized(pre, post, path), None),
            ).with_context(|| "IR15 StrictStaff adoption")?;
        } else { render_state.decline_transition(root, receipts); }
        trial
    } else {
        StrictStaffOverfullCoverageNormalizationTrial {
            source_label: "keine finale Render-XML-Quelle".to_string(),
            verdict: "strict_staff_local_release_skipped_no_render_source".to_string(),
            done_fragment: "StrictStaffOverfullCoverageNormalization: uebersprungen; keine finale Render-XML-Quelle; WriteAllowed nein; MusicMutation nein".to_string(),
            pipeline_line: format!("StrictStaffOverfullCoverageNormalization v{}: uebersprungen; keine finale Render-XML-Quelle; CrossStaffGluingUsedAsReleaseProof 0; WriteAllowed nein; MusicMutation nein.", env!("CARGO_PKG_VERSION")),
            wav_gate_mode: "transient_attack_only_no_sustain_energy_no_overtone_block".to_string(),

            );

            } else { render_state.decline_transition(root, receipts); }

        } else {

            source_label: "keine finale Render-XML-Quelle".to_string(),
            verdict: "final_visible_rest_role_purge_and_vertical_placement_write_skipped_no_render_source".to_string(),
            pipeline_line: format!("RestGlyphVisualEnvelopeStandardFix v{}: uebersprungen; keine finale Render-XML-Quelle; kein Write.", env!("CARGO_PKG_VERSION")),
            done_fragment: "RestGlyphVisualEnvelopeStandardFix: uebersprungen, keine Render-XML-Quelle".to_string(),

fn complete_initial_render_transaction_stage(
    mut best: SolverCandidate,
    mut transaction: InitialRenderTransaction,
    render_state: structural_write_authority::RenderSovereignState,
) -> anyhow::Result<InitialRenderTransactionStageOutcome> {
    let final_export_xml = &transaction.final_export_xml;
    let final_chord_original_evidence =
        frozen_final_chord_original_evidence(&best.material_lineage)?;
    let (final_chord_structural_plan, final_chord_sounding_plan) =
        crate::structural_write_authority::plan_canonical_final_chord_structure_once(
            final_export_xml,
            &final_chord_original_evidence,
        )
        .with_context(|| {
            "FinalChordEncodingSoundingMaterialization konnte kanonischen FinalChord-Plan nicht bilden"
        })?;
    let final_chord_sounding_projection_result =
        crate::structural_write_authority::project_final_chord_encoding_plan_with_provenance(
            final_export_xml,
            &final_chord_sounding_plan,
        )
        .with_context(|| {
            "FinalChordEncodingSoundingMaterialization konnte reine FinalChord-Projektion nicht bilden"
        })?;
    let final_chord_sounding_projection = final_chord_sounding_projection_result.xml;
    let final_chord_source_notes = parse_atomic_notes(final_export_xml);
    let mut final_chord_source_by_range = FinalChordSourceByRange::new();
    let mut next_raw_by_measure = std::collections::BTreeMap::<usize, usize>::new();
    for note in &final_chord_source_notes {
        let measure = note.measure.parse::<usize>().map_err(|_| {
            anyhow!(
                "FinalChordEncodingSoundingMaterialization: non-numeric measure in FinalExport"
            )
        })?;
        let raw_note_index = next_raw_by_measure.entry(measure).or_default();
        if final_chord_source_by_range
            .insert(
                (note.note_start, note.note_end),
                (
                    measure,
                    *raw_note_index,
                    note.start,
                    note.pitch.as_ref().map(|pitch| pitch.midi),
                ),
            )
            .is_some()
        {
            return Err(anyhow!(
                "FinalChordEncodingSoundingMaterialization: source note range is not unique"
            ));
        }
        *raw_note_index = raw_note_index.saturating_add(1);
    }
    if final_chord_source_notes.len() != best.material_lineage.fragments.len()
        || final_chord_source_by_range.len() != best.material_lineage.fragments.len()
    {
        return Err(anyhow!(
            "FinalChordEncodingSoundingMaterialization: FinalExport note topology differs from CandidateMaterial fragments"
        ));
    }

    let mut fragment_by_raw =
        std::collections::BTreeMap::<(usize, usize), CandidateMaterialFragmentState>::new();
    let mut fragment_id_by_raw =
        std::collections::BTreeMap::<(usize, usize), usize>::new();
    for (fragment_index, fragment) in best.material_lineage.fragments.iter().enumerate() {
        let key = (fragment.measure, fragment.raw_note_index);
        if fragment_by_raw.insert(key, fragment.clone()).is_some()
            || fragment_id_by_raw
                .insert(key, fragment_index.saturating_add(1))
                .is_some()
        {
            return Err(anyhow!(
                "FinalChordEncodingSoundingMaterialization: CandidateMaterial raw-note topology is not unique"
            ));
        }
    }

    let mut render_pruned_fragment_ids = Vec::new();
    if transaction.render_xml_pruned_pdf_trial.pdf_source_selected {
        let split_survivor_raw = transaction
            .render_xml_pruned_pdf_trial
            .p2_split_occurrences
            .iter()
            .map(|row| (row.scope, row.raw_note_index))
            .collect::<std::collections::BTreeSet<_>>();
        if split_survivor_raw.len()
            != transaction
                .render_xml_pruned_pdf_trial
                .p2_split_occurrences
                .len()
            || transaction
                .render_xml_pruned_pdf_trial
                .p2_split_occurrences
                .iter()
                .any(|row| {
                    fragment_by_raw
                        .get(&(row.scope, row.raw_note_index))
                        .is_none_or(|fragment| {
                            fragment.onset != row.onset
                                || fragment.pitch != Some(row.pitch)
                                || fragment.staff != row.staff
                                || fragment.is_chord != row.chord_syntax
                        })
                })
        {
            return Err(anyhow!(
                "FinalChordRenderPruneDisposition: split survivor receipt does not match CandidateMaterial identity"
            ));
        }
        render_pruned_fragment_ids = fragment_by_raw
            .iter()
            .filter_map(|(key, fragment)| {
                (fragment.pitch.is_some() && !split_survivor_raw.contains(key))
                    .then(|| fragment_id_by_raw.get(key).copied().map(|id| (id, fragment)))
                    .flatten()
            })
            .collect::<Vec<_>>();
        if render_pruned_fragment_ids.len()
            != transaction
                .render_xml_pruned_pdf_trial
                .removed_safe_hidden_inner_chord_tones
            || render_pruned_fragment_ids
                .iter()
                .any(|(_, fragment)| fragment.visible || !fragment.is_chord)
        {
            return Err(anyhow!(
                "FinalChordRenderPruneDisposition: split receipt does not match the accepted hidden inner-chord prune"
            ));
        }
    }

    let attack_by_source = best
        .material_lineage
        .attacks
        .iter()
        .map(|attack| (attack.source_event_id, attack.clone()))
        .collect::<std::collections::BTreeMap<_, _>>();
    let mut final_chord_source_bindings =
        std::collections::BTreeMap::<(usize, usize), FinalChordSourceBindingRow>::new();
    for provenance in &final_chord_sounding_projection_result.provenance {
        let binding = final_chord_source_binding(
            &mut final_chord_source_bindings,
            (provenance.source_note_start, provenance.source_note_end),
            &final_chord_source_by_range,
            &fragment_by_raw,
            &fragment_id_by_raw,
            "FinalChordProjectionIdentityTransport: projection provenance source range is absent from FinalExport",
            "FinalChordProjectionIdentityTransport: projection provenance source note lost CandidateMaterial fragment",
        )?;
        let source_pitch = binding.source_pitch.ok_or_else(|| {
            anyhow!(
                "FinalChordProjectionIdentityTransport: projection provenance references non-pitched source note"
            )
        })?;
        let fragment = &binding.fragment;
        if fragment.onset != binding.source_onset || fragment.pitch != Some(source_pitch) {
            return Err(anyhow!(
                "FinalChordProjectionIdentityTransport: source topology disagrees with CandidateMaterial fragment"
            ));
        }
        let source_event_id = fragment.source_event_id.ok_or_else(|| {
            anyhow!(
                "FinalChordProjectionIdentityTransport: source fragment has no SourceEventId"
            )
        })?;
        let attack_id = fragment.attack_id.ok_or_else(|| {
            anyhow!(
                "FinalChordProjectionIdentityTransport: source fragment has no SoundingAttackId"
            )
        })?;
        let attack = attack_by_source.get(&source_event_id).cloned().ok_or_else(|| {
            anyhow!(
                "FinalChordProjectionIdentityTransport: SourceEventId has no CandidateMaterial attack"
            )
        })?;
        if attack.attack_id != attack_id || attack.pitch != source_pitch {
            return Err(anyhow!(
                "FinalChordProjectionIdentityTransport: source/attack receipt disagrees with CandidateMaterialState for SourceEventId {}; fragment_id={} measure={} raw_note_index={} source_onset={} fragment_onset={} attack_onset={} source_pitch={} fragment_pitch={:?} attack_pitch={} fragment_attack_id={} state_attack_id={} attack_exists={} fragment_tie_stop={}",
                source_event_id.0,
                binding.fragment_id,
                binding.measure,
                binding.raw_note_index,
                binding.source_onset,
                fragment.onset,
                attack.onset,
                source_pitch,
                fragment.pitch,
                attack.pitch,
                attack_id.0,
                attack.attack_id.0,
                attack.exists,
                fragment.tie_stop,
            ));
        }
    }
    let mut seen_source_ids = std::collections::BTreeSet::<MaterialSourceEventId>::new();
    let mut seen_attack_ids = std::collections::BTreeSet::<SoundingAttackId>::new();
    let mut final_chord_onset_translations =
        Vec::<CandidateMaterialForwardOnsetTranslation>::new();

    for provenance in final_chord_sounding_plan.forward_materializations() {
        let binding = final_chord_source_binding(
            &mut final_chord_source_bindings,
            (provenance.source_note_start, provenance.source_note_end),
            &final_chord_source_by_range,
            &fragment_by_raw,
            &fragment_id_by_raw,
            "FinalChordEncodingSoundingMaterialization: planner provenance source range is absent from FinalExport",
            "FinalChordEncodingSoundingMaterialization: provenance raw note lost CandidateMaterial fragment",
        )?;
        if binding.source_onset != provenance.source_onset {
            return Err(anyhow!(
                "FinalChordEncodingSoundingMaterialization: planner source onset disagrees with FinalExport"
            ));
        }
        let source_pitch = binding.source_pitch.ok_or_else(|| {
            anyhow!(
                "FinalChordEncodingSoundingMaterialization: planner rematerialized a non-pitched source note"
            )
        })?;
        let fragment = &binding.fragment;
        if fragment.onset != binding.source_onset || fragment.pitch != Some(source_pitch) {
            return Err(anyhow!(
                "FinalChordEncodingSoundingMaterialization: forward source topology disagrees with CandidateMaterial fragment"
            ));
        }
        let source_event_id = fragment.source_event_id.ok_or_else(|| {
            anyhow!(
                "FinalChordEncodingSoundingMaterialization: provenance source fragment has no SourceEventId"
            )
        })?;
        let attack = attack_by_source.get(&source_event_id).cloned().ok_or_else(|| {
            anyhow!(
                "FinalChordEncodingSoundingMaterialization: provenance SourceEventId has no SoundingAttack"
            )
        })?;
        let Some(translation) = final_chord_forward_attack_translation(
            &binding, &attack, provenance.source_onset, provenance.projected_sounding_onset,
        )? else {
            continue;
        };
        if !seen_source_ids.insert(source_event_id) || !seen_attack_ids.insert(attack.attack_id) {
            return Err(anyhow!(
                "FinalChordEncodingSoundingMaterialization: planner provenance is not one-to-one in opaque identity"
            ));
        }
        final_chord_onset_translations.push(translation);
    }

    reconcile_candidate_material_forward_onset_translations(
        &mut best,
        "FinalChordEncodingSoundingMaterialization",
        CandidateMaterialDeclaredEffects::final_chord_sounding_materialization(),
        &final_chord_onset_translations,
    )?;

    if transaction.final_render_source_available
        && !transaction
            .p2_assertive_return_write_audit
            .occurrence_receipt_complete
    {
        return Err(anyhow!(
            "RetentionFirstP2OccurrenceContinuation: P2 boundary did not produce a total retained+generated occurrence receipt"
        ));
    }
    let mut p2_render_occurrence_witnesses =
        Vec::<crate::structural_write_authority::FinalChordRenderOccurrenceWitness>::new();
    for receipt in &transaction
        .p2_assertive_return_write_audit
        .generated_occurrences
    {
        let binding = final_chord_source_binding(
            &mut final_chord_source_bindings,
            (receipt.source_note_start, receipt.source_note_end),
            &final_chord_source_by_range,
            &fragment_by_raw,
            &fragment_id_by_raw,
            "RetentionFirstP2OccurrenceContinuation: P2 source range is absent from FinalExport",
            "RetentionFirstP2OccurrenceContinuation: P2 source lost CandidateMaterial fragment",
        )?;
        if binding.measure != receipt.measure {
            return Err(anyhow!(
                "RetentionFirstP2OccurrenceContinuation: generated occurrence measure disagrees with source provenance"
            ));
        }
        let fragment = &binding.fragment;
        let source_event_id = fragment.source_event_id.ok_or_else(|| {
            anyhow!(
                "RetentionFirstP2OccurrenceContinuation: P2 source fragment has no SourceEventId"
            )
        })?;
        let attack_id = fragment.attack_id.ok_or_else(|| {
            anyhow!(
                "RetentionFirstP2OccurrenceContinuation: P2 source fragment has no SoundingAttackId"
            )
        })?;
        p2_render_occurrence_witnesses.push(
            crate::structural_write_authority::FinalChordRenderOccurrenceWitness {
                fragment_id: binding.fragment_id,
                source_event_id: source_event_id.0,
                attack_id: attack_id.0,
                measure: binding.measure,
                pitched_ordinal: receipt.pitched_ordinal,
                representation_duration: None,
                sequence_fingerprint_fnv64: receipt.sequence_fingerprint_fnv64,
            },
        );
    }

    if transaction.final_render_source_available {
        // Continue a participating P2 boundary even without a new P2 write.
        let retained_by_input = transaction
            .p2_assertive_return_write_audit
            .retained_occurrences
            .iter()
            .map(|row| ((row.input_scope, row.input_pitched_ordinal), row))
            .collect::<std::collections::BTreeMap<_, _>>();
        if retained_by_input.len()
            != transaction
                .p2_assertive_return_write_audit
                .retained_occurrences
                .len()
        {
            return Err(anyhow!(
                "RetentionFirstP2OccurrenceContinuation: retained writer receipt reused one input occurrence"
            ));
        }
        if retained_by_input.len()
            != transaction
                .render_xml_pruned_pdf_trial
                .p2_split_occurrences
                .len()
        {
            return Err(anyhow!(
                "RetentionFirstP2OccurrenceContinuation: pitched topology changed between the initial split and P2 input initial_split={} p2_input={}",
                transaction
                    .render_xml_pruned_pdf_trial
                    .p2_split_occurrences
                    .len(),
                retained_by_input.len()
            ));
        }

        let mut consumed_retained_occurrences =
            std::collections::BTreeSet::<(usize, usize)>::new();
        for split_row in &transaction
            .render_xml_pruned_pdf_trial
            .p2_split_occurrences
        {
            let receipt = retained_by_input
                .get(&(split_row.scope, split_row.pitched_ordinal))
                .copied()
                .ok_or_else(|| {
                    anyhow!(
                        "RetentionFirstP2OccurrenceContinuation: split occurrence has no retained P2 writer receipt"
                    )
                })?;
            if !consumed_retained_occurrences
                .insert((split_row.scope, split_row.pitched_ordinal))
            {
                return Err(anyhow!(
                    "RetentionFirstP2OccurrenceContinuation: split occurrence was consumed twice"
                ));
            }
            // The initial split owns CandidateMaterial semantic time; the P2 input receipt
            // owns the current render occurrence after authorized representation-only cursor/rest
            // rewrites. Do not make the later render cursor a second semantic-onset authority.
            // Stable retained-occurrence identity here is scope + input ordinal + pitch + chord
            // syntax; exact persisted range and writer-sequence fingerprint are proven below.
            if receipt.input_scope != split_row.scope
                || receipt.input_pitched_ordinal != split_row.pitched_ordinal
                || receipt.input_pitch != split_row.pitch
                || receipt.input_chord_syntax != split_row.chord_syntax
                || receipt.measure != split_row.scope
            {
                return Err(anyhow!(
                    "RetentionFirstP2OccurrenceContinuation: occurrence topology drifted between the initial split and P2 writer; split(scope={}, raw={}, ordinal={}, onset={}, pitch={}, chord={}) receipt(input_scope={}, input_ordinal={}, input_onset={}, input_pitch={}, input_chord={}, output_measure={}, output_ordinal={})",
                    split_row.scope,
                    split_row.raw_note_index,
                    split_row.pitched_ordinal,
                    split_row.onset,
                    split_row.pitch,
                    split_row.chord_syntax,
                    receipt.input_scope,
                    receipt.input_pitched_ordinal,
                    receipt.input_onset,
                    receipt.input_pitch,
                    receipt.input_chord_syntax,
                    receipt.measure,
                    receipt.pitched_ordinal,
                ));
            }
            let fragment = fragment_by_raw
                .get(&(split_row.scope, split_row.raw_note_index))
                .cloned()
                .ok_or_else(|| {
                    anyhow!(
                        "RetentionFirstP2OccurrenceContinuation: split occurrence lost CandidateMaterial fragment"
                    )
                })?;
            let fragment_id = fragment_id_by_raw
                .get(&(split_row.scope, split_row.raw_note_index))
                .copied()
                .ok_or_else(|| {
                    anyhow!(
                        "RetentionFirstP2OccurrenceContinuation: split occurrence lost canonical fragment id"
                    )
                })?;
            if fragment.measure != split_row.scope
                || fragment.raw_note_index != split_row.raw_note_index
                || fragment.onset != split_row.onset
                || fragment.pitch != Some(split_row.pitch)
                || fragment.staff != split_row.staff
                || fragment.is_chord != split_row.chord_syntax
            {
                return Err(anyhow!(
                    "RetentionFirstP2OccurrenceContinuation: split locator disagrees with CandidateMaterial identity"
                ));
            }
            let source_event_id = fragment.source_event_id.ok_or_else(|| {
                anyhow!(
                    "RetentionFirstP2OccurrenceContinuation: split fragment has no SourceEventId"
                )
            })?;
            let attack_id = fragment.attack_id.ok_or_else(|| {
                anyhow!(
                    "RetentionFirstP2OccurrenceContinuation: split fragment has no SoundingAttackId"
                )
            })?;
            p2_render_occurrence_witnesses.push(
                crate::structural_write_authority::FinalChordRenderOccurrenceWitness {
                    fragment_id,
                    source_event_id: source_event_id.0,
                    attack_id: attack_id.0,
                    measure: receipt.measure,
                    pitched_ordinal: receipt.pitched_ordinal,
                    representation_duration: None,
                    sequence_fingerprint_fnv64: receipt.sequence_fingerprint_fnv64,
                },
            );
        }
    }

    let witnessed_fragment_ids = p2_render_occurrence_witnesses
        .iter()
        .map(|witness| witness.fragment_id)
        .collect::<std::collections::BTreeSet<_>>();
    transaction.render_pruned_final_chord_omissions = render_pruned_fragment_ids
        .drain(..)
        .map(|(fragment_id, _)| fragment_id)
        .filter(|fragment_id| !witnessed_fragment_ids.contains(fragment_id))
        .collect();

    let mut provenance_by_projected_start = std::collections::BTreeMap::new();
    for provenance in &final_chord_sounding_projection_result.provenance {
        if provenance_by_projected_start
            .insert(provenance.projected_note_start, provenance)
            .is_some()
        {
            return Err(anyhow!(
                "NotationVoicePreSerializationConstraintClosureAudit: projected pitched-note provenance start is not unique"
            ));
        }
    }
    let projected_notes = parse_atomic_notes(&final_chord_sounding_projection);
    let mut next_pitched_ordinal_by_measure =
        std::collections::BTreeMap::<usize, usize>::new();
    let mut atomic_group_tracker =
        crate::audit_only::notation_voice_pre_serialization_constraint_closure_audit::
            PreSerializationAtomicGroupTracker::default();
    let mut pre_serialization_occurrences = Vec::<
        crate::audit_only::notation_voice_pre_serialization_constraint_closure_audit::
            PreSerializationOccurrenceRow,
    >::new();
    let mut pre_serialization_occupancy = Vec::<
        crate::audit_only::notation_voice_pre_serialization_constraint_closure_audit::
            PreSerializationVisibleOccupancyRow,
    >::new();
    for note in &projected_notes {
        let measure = note.measure.parse::<usize>().map_err(|_| {
            anyhow!(
                "NotationVoicePreSerializationConstraintClosureAudit: projected FinalChord note has non-numeric measure"
            )
        })?;
        let staff = note.staff.parse::<u8>().map_err(|_| {
            anyhow!(
                "NotationVoicePreSerializationConstraintClosureAudit: projected FinalChord note has non-numeric staff"
            )
        })?;
        if !matches!(staff, 1 | 2) {
            return Err(anyhow!(
                "NotationVoicePreSerializationConstraintClosureAudit: projected FinalChord note has unsupported staff {}",
                staff
            ));
        }
        let block = final_chord_sounding_projection
            .get(note.note_start..note.note_end)
            .ok_or_else(|| {
                anyhow!(
                    "NotationVoicePreSerializationConstraintClosureAudit: projected FinalChord note range is invalid"
                )
            })?;
        let is_rest = block.contains("<rest");
        let (atomic_group_id, atomic_group_kind) = atomic_group_tracker.assign(
            measure,
            note.visible,
            is_rest,
            note.is_chord,
            note.pitch.is_some(),
        );
        let mut source_event_id = None;
        let mut attack_id = None;
        if let Some(pitch) = note.pitch.as_ref() {
            let provenance = provenance_by_projected_start
                .get(&note.note_start)
                .copied()
                .ok_or_else(|| {
                    anyhow!(
                        "NotationVoicePreSerializationConstraintClosureAudit: projected pitched note has no exact FinalChord provenance"
                    )
                })?;
            let binding = final_chord_source_bindings
                .get(&(provenance.source_note_start, provenance.source_note_end))
                .ok_or_else(|| {
                    anyhow!(
                        "NotationVoicePreSerializationConstraintClosureAudit: FinalChord provenance lost source binding"
                    )
                })?;
            let source_id = binding.fragment.source_event_id.ok_or_else(|| {
                anyhow!(
                    "NotationVoicePreSerializationConstraintClosureAudit: pitched source binding has no SourceEventId"
                )
            })?;
            let sounding_attack_id = binding.fragment.attack_id.ok_or_else(|| {
                anyhow!(
                    "NotationVoicePreSerializationConstraintClosureAudit: pitched source binding has no SoundingAttackId"
                )
            })?;
            source_event_id = Some(source_id.0);
            attack_id = Some(sounding_attack_id.0);
            let pitched_ordinal = next_pitched_ordinal_by_measure.entry(measure).or_default();
            pre_serialization_occurrences.push(
                crate::audit_only::notation_voice_pre_serialization_constraint_closure_audit::
                    PreSerializationOccurrenceRow {
                        source_event_id: source_id.0,
                        attack_id: sounding_attack_id.0,
                        measure,
                        pitched_ordinal: *pitched_ordinal,
                        represented_onset: note.start,
                        represented_duration: note.duration,
                        pitch_midi: pitch.midi,
                        staff,
                        current_voice: note.voice.clone(),
                    },
            );
            *pitched_ordinal = (*pitched_ordinal).saturating_add(1);
        }
        pre_serialization_occupancy.push(
            crate::audit_only::notation_voice_pre_serialization_constraint_closure_audit::
                PreSerializationVisibleOccupancyRow {
                    measure,
                    staff,
                    represented_onset: note.start,
                    represented_duration: note.duration,
                    visible: note.visible,
                    is_rest,
                    is_chord_member: note.is_chord,
                    pitch_midi: note.pitch.as_ref().map(|pitch| pitch.midi),
                    source_event_id,
                    attack_id,
                    atomic_group_id,
                    atomic_group_kind,
                },
        );
    }

    freeze_final_sounding_material(
        &mut best,
        &final_chord_sounding_projection,
        final_export_xml,
        &final_chord_structural_plan,
        &pre_serialization_occurrences,
        &pre_serialization_occupancy,
    )?;
    crate::structural_write_authority::attach_final_chord_render_occurrence_witnesses(
        &mut best
            .material_lineage
            .frozen_sounding_temporal_lane_plan
            .final_chord_structural_plan,
        p2_render_occurrence_witnesses,
    )
    .with_context(|| {
        "RetentionFirstP2OccurrenceContinuation konnte Writer-Receipt nicht an den kanonischen FinalChord-Plan binden"
    })?;
    transaction.frozen_sounding_temporal_lane_plan =
        best.material_lineage.frozen_sounding_temporal_lane_plan.clone();
    transaction.frozen_sounding_material_attacks = best.material_lineage.attacks.clone();

    Ok(InitialRenderTransactionStageOutcome {
        best,
        transaction,
        render_state,
    })
}
