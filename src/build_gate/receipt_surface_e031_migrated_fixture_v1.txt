fn add_tenuto_to_note_fragment_authorized(source: &str) -> (Option<String>, Option<AdditiveTransformationReceipt>) {
    let geometry = bound_tenuto_fragment_insertions(source);
    let output = (|| {
    if source.contains("<tenuto") {
        return Some(source.to_string());
    }
    let token = TenutoCapability {
        source: ExactSourceBinding::issue(source),
    };
    let mut out = token.source.into_verified_string(source)?;
    if let Some(start) = out.find("<articulations>") {
        out.insert_str(start + "<articulations>".len(), "\n        <tenuto/>");
        return Some(out);
    }
    if let Some(position) = out.rfind("</notations>") {
        out.insert_str(
            position,
            "      <articulations>\n        <tenuto/>\n      </articulations>\n",
        );
        return Some(out);
    }
    let position = out.rfind("</note>")?;
    out.insert_str(
        position,
        "      <notations>\n        <articulations>\n          <tenuto/>\n        </articulations>\n      </notations>\n",
    );
    Some(out)
})();
    let receipt = crate::master_state_identity::AdditiveTransformationReceipt::mint_tenuto_fragment_receipt(super::TenutoFragmentAdditiveMint, source, output.as_deref().unwrap_or(source), geometry);
    assert!(output.as_deref().is_none_or(|post| post == source) || receipt.is_some(), "fragment owner changed bytes without additive receipt");
    (output, receipt)
}
